cmake_minimum_required(VERSION 3.12)
set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
project(base-rpc VERSION 0.1.0 LANGUAGES C CXX)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  set(BASE_RPC_TOP_LEVEL ON)
else()
  set(BASE_RPC_TOP_LEVEL OFF)
endif()
option(BASE_RPC_BUILD_TESTS "Build base-rpc tests" ${BASE_RPC_TOP_LEVEL})
option(BASE_RPC_BUILD_EXAMPLES "Build base-rpc examples" ${BASE_RPC_TOP_LEVEL})

if(BASE_RPC_TOP_LEVEL)
  if(NOT CMAKE_BUILD_TYPE OR CMAKE_BUILD_TYPE STREQUAL "None")
    set(CMAKE_BUILD_TYPE "RelWithDebInfo")
  endif()
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_POSITION_INDEPENDENT_CODE ON)
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fcoroutines")
  elseif(CMAKE_CXX_COMPILER_ID STREQUAL "Clang" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 17)
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fcoroutines-ts")
  endif()
  # folly's F14 containers pick their implementation from these flags, so folly, base-rpc and users must agree
  # (3FS builds everything with them). A parent project embedding base-rpc has to do the same.
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64")
    add_compile_options(-msse4.2 -mavx2)
  endif()
  include(cmake/Sanitizers.cmake)
endif()
include(cmake/CompileFlags.cmake)

find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS filesystem system program_options)
find_path(IBVERBS_INCLUDE_DIR infiniband/verbs.h REQUIRED)
find_library(IBVERBS_LIBRARY ibverbs REQUIRED)

# Third party libs. Skip the ones the parent project already provides (it must provide the patched folly).
store_compile_flags()
if(NOT TARGET fmt::fmt AND NOT TARGET fmt)
  add_subdirectory(third_party/fmt EXCLUDE_FROM_ALL)
endif()
if(NOT TARGET libzstd_static)
  set(ZSTD_BUILD_STATIC ON)
  add_subdirectory(third_party/zstd/build/cmake EXCLUDE_FROM_ALL)
  set(zstd_INCLUDE_DIRS "${PROJECT_SOURCE_DIR}/third_party/zstd/lib")
  set(ZSTD_INCLUDE_DIR "${PROJECT_SOURCE_DIR}/third_party/zstd/lib")
  set(zstd_FOUND ON)
  set(ZSTD_FOUND ON)
  set(zstd_LIBRARIES "${CMAKE_BINARY_DIR}/third_party/zstd/build/cmake/lib/libzstd.a")
  set(ZSTD_LIBRARY "${zstd_LIBRARIES}")
endif()
restore_compile_flags()
if(BASE_RPC_BUILD_TESTS AND NOT TARGET gtest)
  add_subdirectory(third_party/googletest EXCLUDE_FROM_ALL)
  restore_compile_flags()
endif()
if(NOT TARGET folly)
  # folly needs 3FS's local patches; apply once (idempotent).
  execute_process(COMMAND git -C ${PROJECT_SOURCE_DIR}/third_party/folly apply --reverse --check
                          ${PROJECT_SOURCE_DIR}/patches/folly.patch RESULT_VARIABLE FOLLY_PATCHED
                  OUTPUT_QUIET ERROR_QUIET)
  if(FOLLY_PATCHED)
    execute_process(COMMAND git -C ${PROJECT_SOURCE_DIR}/third_party/folly apply ${PROJECT_SOURCE_DIR}/patches/folly.patch
                    RESULT_VARIABLE FOLLY_PATCH_FAILED)
    if(FOLLY_PATCH_FAILED)
      message(FATAL_ERROR "failed to apply patches/folly.patch to third_party/folly")
    endif()
  endif()
  set(FOLLY_NO_EXCEPTION_TRACER ON)
  add_subdirectory(third_party/folly EXCLUDE_FROM_ALL)
  restore_compile_flags()
endif()
if(NOT TARGET scn::scn)
  set(SCN_TESTS OFF)
  set(SCN_EXAMPLES OFF)
  set(SCN_BENCHMARKS OFF)
  set(SCN_DOCS OFF)
  set(SCN_INSTALL OFF)
  set(SCN_PEDANTIC OFF)
  add_subdirectory(third_party/scnlib EXCLUDE_FROM_ALL)
  restore_compile_flags()
endif()
if(NOT TARGET toml11)
  add_subdirectory(third_party/toml11 EXCLUDE_FROM_ALL)
  restore_compile_flags()
endif()

file(GLOB_RECURSE BASE_RPC_SOURCES CONFIGURE_DEPENDS ${PROJECT_SOURCE_DIR}/src/*.cc)
add_library(base-rpc STATIC ${BASE_RPC_SOURCES})
target_include_directories(base-rpc PUBLIC ${PROJECT_SOURCE_DIR}/src ${PROJECT_SOURCE_DIR} ${PROJECT_SOURCE_DIR}/compat ${IBVERBS_INCLUDE_DIR})  # root: ZSTD.h includes "third_party/zstd/lib/zstd.h"
target_link_libraries(base-rpc PUBLIC folly scn::scn toml11 libzstd_static ${IBVERBS_LIBRARY} Threads::Threads)
target_compile_options(base-rpc PRIVATE -Wall -Wextra)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  # Serde.cc keeps 3FS's `template <> template <>` form for clang, which clang >= 19 reports as an error by default.
  target_compile_options(base-rpc PRIVATE -Wno-extraneous-template-head)
endif()

if(BASE_RPC_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(BASE_RPC_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()
